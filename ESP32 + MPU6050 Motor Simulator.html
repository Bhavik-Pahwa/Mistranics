Arduino code for the real Nano ESP32 / ESP32

#include <Wire.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>

Adafruit_MPU6050 mpu;
const float WARN_G = 0.15, FAULT_G = 0.35, WARN_T = 60, FAULT_T = 75;
const int N = 200;
float bx[N], by[N], bz[N];
int idx = 0;
unsigned long lastPrint = 0;

float rmsOf(float *b) {
  float m = 0; for (int i = 0; i < N; i++) m += b[i]; m /= N;
  float s = 0; for (int i = 0; i < N; i++) s += (b[i] - m) * (b[i] - m);
  return s / N;                         // variance
}

void setup() {
  Serial.begin(115200);
  Wire.begin();                         // Nano ESP32: A4 = SDA, A5 = SCL
  if (!mpu.begin()) { Serial.println("MPU6050 not found"); while (1) delay(10); }
  mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
  mpu.setGyroRange(MPU6050_RANGE_500_DEG);
  mpu.setFilterBandwidth(MPU6050_BAND_260_HZ);
}

void loop() {
  sensors_event_t a, g, t;
  mpu.getEvent(&a, &g, &t);
  bx[idx] = a.acceleration.x / 9.80665;
  by[idx] = a.acceleration.y / 9.80665;
  bz[idx] = a.acceleration.z / 9.80665;
  idx = (idx + 1) % N;

  if (millis() - lastPrint > 3000) {   // print every 3000 ms
    lastPrint = millis();
    float rms = sqrt(rmsOf(bx) + rmsOf(by) + rmsOf(bz));
    const char *st = (rms > FAULT_G || t.temperature > FAULT_T) ? "FAULT"
                   : (rms > WARN_G  || t.temperature > WARN_T)  ? "WARNING" : "NORMAL";
    Serial.printf("ax=%.2f ay=%.2f az=%.2f gz=%.1f T=%.1f rms=%.3f %s\n",
      bx[(idx + N - 1) % N], by[(idx + N - 1) % N], bz[(idx + N - 1) % N],
      g.gyro.z * 57.2958, t.temperature, rms, st);
  }
  delay(3000);                             // one reading every 3 seconds
}
